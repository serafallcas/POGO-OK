-- Readable consolidated view for UI and reporting
CREATE OR REPLACE VIEW question_expert_guidance AS
SELECT
  q.id AS question_id,
  q.question_code,
  q.question_text,
  q.domain_id,
  q.process_id,
  q.primary_dimension,
  q.evidence_min_required,
  q.expert_review_status,
  q.expert_basis,
  d.label AS domain_label,
  spa.label AS sub_process_area_label,
  p.label AS process_label,
  ig.objective,
  ig.assessment_scope,
  ig.interview_steps,
  ig.expected_evidence_summary,
  ig.validation_method,
  ig.scoring_guidance,
  ig.red_flags AS instruction_red_flags,
  ig.good_practice_signals,
  ig.alert_rule,
  eg.expected_evidence_detail,
  eg.validation_checks,
  eg.red_flags AS evidence_red_flags,
  eg.minimum_evidence_count,
  eg.preferred_evidence_count,
  eg.evidence_quality_rule,
  eg.acceptable_formats,
  eg.recency_requirement,
  eg.mandatory_for_score_at_or_above,
  mg.level_1_basic,
  mg.level_2_developing,
  mg.level_3_established,
  mg.level_4_advanced,
  mg.level_5_leading,
  mg.scoring_rule
FROM questions q
LEFT JOIN domains d ON d.id = q.domain_id
LEFT JOIN processes p ON p.id = q.process_id
LEFT JOIN sub_process_areas spa ON spa.id = p.sub_process_area_id
LEFT JOIN question_instruction_guides ig ON ig.question_id = q.id
LEFT JOIN question_evidence_guides eg ON eg.question_id = q.id
LEFT JOIN question_maturity_guides mg ON mg.question_id = q.id;

-- RLS helper for reference-data administration
CREATE OR REPLACE FUNCTION is_pumma_reference_admin()
RETURNS boolean
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
  SELECT EXISTS (
    SELECT 1
    FROM profiles p
    WHERE p.auth_user_id = auth.uid()
      AND p.status = 'active'
      AND p.role IN ('super_admin', 'program_admin')
  );
$$;

-- Replace legacy unrestricted write policies on reference data.
-- Read access remains available to authenticated users; writes require reference-admin privileges.
DO $$
DECLARE
  ref_table text;
  policy_name text;
BEGIN
  FOREACH ref_table IN ARRAY ARRAY[
    'evaluation_types', 'domains', 'processes', 'maturity_models',
    'maturity_levels', 'maturity_statements', 'questionnaires',
    'questions', 'alert_rules'
  ]
  LOOP
    FOREACH policy_name IN ARRAY ARRAY[
      'auth_insert_' || ref_table,
      'auth_update_' || ref_table,
      'auth_delete_' || ref_table
    ]
    LOOP
      EXECUTE format('DROP POLICY IF EXISTS %I ON %I', policy_name, ref_table);
    END LOOP;

    EXECUTE format('DROP POLICY IF EXISTS %I ON %I', 'admin_write_' || ref_table, ref_table);
    EXECUTE format(
      'CREATE POLICY %I ON %I FOR ALL TO authenticated USING (is_pumma_reference_admin()) WITH CHECK (is_pumma_reference_admin())',
      'admin_write_' || ref_table,
      ref_table
    );
  END LOOP;
END $$;
